Write-Host "Simula ng pag-install ng Python at mga libraries..." -ForegroundColor Cyan

# 1. I-download at i-install ang Official Python sa System (May kasamang pip at PATH)
$pyInstallerUrl = "https://www.python.org/ftp/python/3.11.9/python-3.11.9-amd64.exe"
$pyInstallerPath = Join-Path $env:TEMP "python_installer.exe"

Write-Host "Dina-download ang Python installer..." -ForegroundColor Cyan
Invoke-WebRequest -Uri $pyInstallerUrl -OutFile $pyInstallerPath

Write-Host "Ini-install ang Python sa system (Tahimik na proseso)..." -ForegroundColor Cyan
Start-Process -FilePath $pyInstallerPath -ArgumentList "/quiet InstallAllUsers=1 PrependPath=1 Include_test=0" -Wait

Remove-Item -Path $pyInstallerPath -Force

# I-refresh ang Environment Variables sa kasalukuyang session
$env:Path = [System.Environment]::GetEnvironmentVariable("Path","Machine") + ";" + [System.Environment]::GetEnvironmentVariable("Path","User")

Write-Host "Tapos na ang pag-install ng Python!" -ForegroundColor Green

# 2. I-install ang mga Libraries (tulad ng pynput)
Write-Host "Ina-update ang pip at ini-install ang pynput..." -ForegroundColor Cyan
python -m pip install --upgrade pip
python -m pip install pynput

Write-Host "Tapos na ang lahat! Handa na ang Python at ang mga libraries sa iyong PC." -ForegroundColor Green
